ឆ្នេរសមុទ្រដ៏ស្រស់ស្អាតនៃកម្ពុជា

សម្រាកលំហែកាយនៅតាមកោះ និងឆ្នេរខ្សាច់ក្បុសនៃខេត្តជាប់មាត់សមុទ្រ។

ទិសដៅកម្សាន្តមាត់សមុទ្រ

កោះរ៉ុងសន្លឹម

ជាកោះដែលមានទឹកសមុទ្រថ្លាយង់ ស្រឡូន និងស័ក្តិសមបំផុតសម្រាប់ការសម្រាកលំហែកាយ។

ខេត្តព្រះសីហនុ

ទីក្រុងមាត់សមុទ្រដែលមានភាពអ៊ូអរ រមណីយដ្ឋាន និងសេវាកម្មទេសចរណ៍សម្បូរបែប។

ខេត្តកំពត និងកែប

ល្បីល្បាញខាងទេសចរណ៍ធម្មជាតិ ភ្នំបូកគោ និងអាហារសមុទ្រស្រស់ៗ (ជាពិសេសក្តាមសមុទ្រ)។
